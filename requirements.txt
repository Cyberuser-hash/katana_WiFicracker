# ============================================================
# DEPENDENCIES FOR KATANA.PY (Wi-Fi Pentesting Tool)
# ============================================================
# Installation: sudo apt-get update && sudo apt-get install -y [packages]
# For Debian/Ubuntu/Kali Linux

# ============================================================
# CORE AIRCRACK-NG SUITE (required)
# ============================================================
aircrack-ng          # airodump-ng, aireplay-ng, airmon-ng

# ============================================================
# HCXTOOLS (for handshake to hashcat format conversion)
# ============================================================
hcxtools             # hcxpcapngtool

# ============================================================
# HASHCAT (for WPA/WPA2 handshake bruteforce)
# ============================================================
hashcat              # bruteforce handshake (mode 22000)

# ============================================================
# TERMINAL EMULATOR (for popup windows)
# ============================================================
xterm                # popup windows for airodump, aireplay, etc.

# ============================================================
# NETWORK MANAGEMENT (for stopping NetworkManager)
# ============================================================
systemd              # systemctl stop NetworkManager
NetworkManager       # network connection management
wpa_supplicant       # for stopping/pkill

# ============================================================
# WIRELESS TOOLS (adapter management)
# ============================================================
wireless-tools       # iwconfig, iwlist
iw                   # iw dev (channel settings)
rfkill               # wifi blocking management
pciutils             # lspci (detect wifi adapter)
usbutils             # lsusb (detect USB adapter)

# ============================================================
# BASE SYSTEM UTILITIES (usually pre-installed)
# ============================================================
iproute2             # ip command (ip link, ip address)
procps               # pkill command
sudo                 # privilege escalation
coreutils            # rm, clear and other basic utilities
